Transport of neutral particles are caused by the ``random walk'' movement of the particles, and because there is no inherent preference for a particular direction, the transported quantity -- being momentum or energy
-- is eventually equilibrated. This is also true in the case of charged particles in the \textit{unmagnetized} limits; that is when $\lVert \mathbf{B} \rVert = 0$. The existence of a magnetic field gives rise to
anisotropy in heat conduction, since the particles are forced to spiral along the magnetic field lines, so the rate of transport is the highest in the direction parallel to the magnetic field lines, i.e.
$\displaystyle \mathbf{\hat{b}} = \frac{\mathbf{\hat{B}}}{\lVert \mathbf{B} \rVert}$. On the other hand, motion across the field lines (the perpendicular direction) are strongly suppressed, so there is little
transport in that direction. The other direction, which is perpendicular to both $\mathbf{\hat{b}}$ and the gradient of the transported quantity and termed \textit{cross} direction in this document, is the result of
the Righi–Leduc effect.

Transport processes in plasmas are pioneered by Braginskii in his 1965 article \cite{Braginskii}, where the transport properties -- namely viscosity and thermal conductivity -- are given as function of magnetization
$\chi$ (also called the Hall parameter). It is defined as
\begin{equation}
    \chi_j \equiv \frac{\omega_{\text{ce}, j}}{\nu_{ji}},\ j = e,i
\end{equation}
and the electron or ion gyrofrequency is
\begin{equation}
    \omega_{\text{ce}, j} \equiv \frac{eB}{m_j}.
\end{equation}

The electron-ion collision frequency is given in Equation \ref{eqn:nu_ei_momentum}, and the ion-ion collision frequency is
\begin{equation}
    \nu_{ii} = \frac{4}{3} \sqrt{\pi} \frac{n_i Z^4 e^4}{(4\pi\varepsilon_0)^2 \sqrt{m (k_B T_i)^3}} \ln\Lambda_i.
\end{equation}

\subsection{Heat flux}
In the presence of a magnetic field, the ion heat flux is as followed:
\begin{equation}
    \mathbf{q}_i'' = -\kappa^i_\parallel \nabla_\parallel T_i - \kappa^i_\perp \nabla_\perp T_i + \kappa^i_\times \nabla_\times T_i
    \label{eqn:ion_heat_flux}
\end{equation}

The three components of $\nabla T_i$,
\begin{subequations}
    \label{eqn:components_of_nablaT}
    \begin{align}
        \nabla_\parallel T_i &= (\mathbf{\hat{b}} \cdot \nabla T_i) \mathbf{\hat{b}} \\
        \nabla_\perp T_i &= \nabla T_e - \nabla_\parallel T_i \\
        \nabla_\times T_i &= \mathbf{\hat{b}} \times \nabla T_i,
    \end{align}
\end{subequations}
represent, in the respective order, the parallel, perpendicular, and cross components of $\nabla T_i$ with respect to the magnetic field direction $\mathbf{\hat{b}}$.

Using the definitions in Equation \ref{eqn:components_of_nablaT}, Equation \ref{eqn:ion_heat_flux} can be written directly in terms of $\nabla T_i$:
\begin{subequations}
    \label{eqn:electron_conduction_cartesian}
    \begin{align}
        \mathbf{q}_i'' &= -\kappa^i_\parallel (\mathbf{\hat{b}} \cdot \nabla T_i) \mathbf{\hat{b}} - \kappa^i_\perp (\nabla T_i - (\mathbf{\hat{b}} \cdot \nabla T_i) \mathbf{\hat{b}}) + \kappa^i_\times \mathbf{\hat{b}} \times \nabla T_i \\
        &= - \underbrace{\left( \kappa^i_\perp \mathbf{I} + (\kappa^i_\parallel - \kappa^i_\perp) \mathbf{\hat{b}}\mathbf{\hat{b}}^T - \kappa^i_\times \mathbf{B}_\times \right)}_{\mathbf{K}_i} \nabla T_i,
    \end{align}
\end{subequations}
where $\mathbf{B}_\times$ is the antisymmetric matrix
\begin{equation}
    \mathbf{B}_\times =
    \begin{bmatrix}
        0 & -b_z & b_y \\
        b_z & 0 & -b_x \\
        -b_y & b_x & 0
    \end{bmatrix},
\end{equation}
that satisfies $\mathbf{B}_\times \mathbf{v} = \mathbf{\hat{b}} \times \mathbf{v}$ for any vector $\mathbf{v}$.

The ion thermal conductivities are given as rational functions of $\chi_i$ in Equation 4.40 of Braginskii (1965) \cite{Braginskii}:
\begin{subequations}
    \label{eqn:kappa_e}
    \begin{align}
        \kappa_\parallel^i &= \hat{\kappa}_i \frac{{\gamma'_0}^i}{\delta_0^i}, \\
        \kappa_\perp^i &= \hat{\kappa}_i \frac{{\gamma'_1}^i\chi_i^2 + \gamma'_0}{\chi_i^4 + \delta_1^i\chi_e^2 + \delta_0^i}, \\
        \kappa_\times^i &= \hat{\kappa}_i \frac{\chi_i({\gamma''_1}\chi_i^2 + \gamma_0'')}{\chi_i^4 + \delta_1^i\chi_e^2 + \delta_0^i},
    \end{align}
    with the prefractor $\hat{\kappa}_i$ defined as
    \begin{equation}
        \hat{\kappa}_i \equiv \frac{n_i k_B^2 T_i}{m_i \nu_{ii}}.
        \label{eqn:kappa_hat_i}
    \end{equation}
\end{subequations}

In the unmagnetized limit ($\chi_i \rightarrow 0$), the parallel and perpendicular $\kappa$s are identical while the cross term $\kappa_\times^i$ vanishes. The isotropic flux is recovered:
\begin{equation}
    \mathbf{q}_i'' = -\kappa_i \nabla T_i
    \label{eqn:isotropic_ion_heat_flux}    
\end{equation}

The electron heat flux has a conductive component similar to that of ion, but also contains a thermoelectric term that is expressed in terms of the current density $\mathbf{j}$:
\begin{subequations}
    \label{eqn:electron_heat_flux}
    \begin{align}
        \mathbf{q}_e'' &= -\kappa^e_\parallel \nabla_\parallel T_e - \kappa^e_\perp \nabla_\perp T_e - \kappa^e_\times \nabla_\times T_e - \frac{k_B T_e}{e} \left( \beta_\parallel \mathbf{j}_\parallel + \beta_\perp \mathbf{j}_\perp + \beta_\times \mathbf{j}_\times \right) \\
        &= -\mathbf{K}_e \nabla T_e - \mathbf{\Pi} \mathbf{j},
    \end{align}
    where the conductivity and thermoelectric flux tensors are
    \begin{align}
        \mathbf{K}_e &\equiv \kappa^e_\perp \mathbf{I} + (\kappa^e_\parallel - \kappa^e_\perp) \mathbf{\hat{b}}\mathbf{\hat{b}}^T + \kappa^e_\times \mathbf{B}_\times \\
        \mathbf{\Pi} &\equiv \frac{k_B T_e}{e} \left[\beta_\perp \mathbf{I} + (\beta_\parallel - \beta_\perp) \mathbf{\hat{b}}\mathbf{\hat{b}}^T + \beta_\times \mathbf{B}_\times \right]
    \end{align}
\end{subequations}

The $\kappa^e$'s and $\beta$'s are given in Table 2 Braginskii (1965), since in general they have a dependency on $Z$, though in this paper $Z = 1$ is assumed \cite{Braginskii}. Another source of electron transport properties is
Epperlein and Haines (1986), and they gives a more accurate expression of perpendicular and cross properties in the moderate $\chi_e$ domain ($0.3 \leq \chi_e \leq 30$) \cite{EpperleinHaines}. Since this article does
not contain the equivalent formulae for ion properties, it is not used in favor of the Braginskii expressions, but is cited here as a reference. The electron conductivities are
\begin{subequations}
    \label{eqn:kappa_e}
    \begin{align}
        \kappa_\parallel^e &= \hat{\kappa}_e \frac{{\gamma'}_0^e}{\delta_0^e}, \\
        \kappa_\perp^e &= \hat{\kappa}_e \frac{{\gamma'}_1^e\chi_e^2 + {\gamma'}_0}{\chi_e^4 + \delta^e_1\chi_e^2 + \delta^e_0}, \\
        \kappa_\times^e &= \hat{\kappa}_e \frac{\chi_e({\gamma''}_1^e\chi_e^2 + {\gamma''}_0^e)}{\chi_e^4 + \delta^e_1\chi_e^2 + \delta^e_0},
    \end{align}
    with the prefractor $\hat{\kappa}_e$ defined as
    \begin{equation}
        \hat{\kappa}_e \equiv \frac{n_e k_B^2 T_e}{m_e \nu_{ei}^M}.
        \label{eqn:kappa_hat_e}
    \end{equation}
\end{subequations}

And the thermoelectric coefficients are
\begin{subequations}
    \label{eqn:beta_e}
    \begin{align}
        \beta_\parallel &= \frac{\beta'_0}{\delta_0}, \\
        \beta_\perp &= \frac{\beta'_1\chi_e^2 + \beta'_0}{\chi_e^4 + \delta^e_1\chi_e^2 + \delta^e_0}, \\
        \beta_\times &= \frac{\chi_e(\beta_1'' \chi_e^2 + \beta_0'')}{\chi_e^4 + \delta^e_1\chi_e^2 + \delta^e_0}.
    \end{align}
\end{subequations}

In the unmagnetized limit ($\chi_e \rightarrow 0$), the electron heat flux is
\begin{equation}
    \mathbf{q}_e'' = -\kappa_e \nabla T_e - \frac{\beta k_B T_e}{e} \mathbf{j}.
    \label{eqn:isotropic_electron_heat_flux}    
\end{equation}

\subsection{Momentum flux}
Braginskii gives the total stress tensor as the sum of five component tensors \cite{Braginskii}:
\begin{subequations}
    \begin{align}
        \mathbf{T} &= \sum_{n=0}^{4} \mathbf{T}_n \\
        \mathbf{T}_0 &= 3\mu_0 \left(\mathbf{\hat{b}}\mathbf{\hat{b}}^T - \frac{1}{3}\mathbf{I} \right) (\mathbf{\hat{b}} \cdot \mathbf{W}\mathbf{\hat{b}}) \\ % \left(\mathbf{\hat{b}}\mathbf{\hat{b}}^T - \frac{1}{3}\mathbf{I} \right) : \nabla \mathbf{u} \\
        \mathbf{T}_1 &= \mu_1 \left[\mathbf{P}\mathbf{W}\mathbf{P} + \frac{1}{2}\mathbf{P} (\mathbf{\hat{b}} \cdot \mathbf{W}\mathbf{\hat{b}}) \right] \\
        \mathbf{T}_2 &= \mu_2 \left[\mathbf{P}\mathbf{W}\mathbf{\hat{b}}\mathbf{\hat{b}}^T + \mathbf{\hat{b}}\mathbf{\hat{b}}^T\mathbf{W}\mathbf{P} \right] \\
        \mathbf{T}_3 &= -\frac{\mu_3}{2} \left[\mathbf{B}_\times\mathbf{W}\mathbf{P} - \mathbf{P}\mathbf{W}\mathbf{B}_\times \right] \\
        \mathbf{T}_4 &= -\mu_4 \left[\mathbf{B}_\times\mathbf{W}\mathbf{\hat{b}}\mathbf{\hat{b}}^T - \mathbf{\hat{b}}\mathbf{\hat{b}}^T \mathbf{W}\mathbf{B}_\times \right],
    \end{align}
    where the projection matrix $\mathbf{P}$ is
    \begin{equation}
        \mathbf{P} \equiv \mathbf{I} - \mathbf{\hat{b}}\mathbf{\hat{b}}^T,
    \end{equation}
    and the rate-of-strain tensor $\mathbf{W}$ is
    \begin{equation}
        \mathbf{W} \equiv \left[\nabla\mathbf{u} + (\nabla\mathbf{u})^T\right] - \frac{2}{3}\mathbf{I}\nabla\cdot\mathbf{u}
    \end{equation}    
\end{subequations}

The viscosities are rational functions of $\chi$, and their polynomial coefficients are given in Equations 4.44 through 4.45 of Braginskii \cite{Braginskii}:
\begin{subequations}
    \label{eqn:mu_j}
    \begin{align}
        \mu_0^j &= \hat{\mu}_j \frac{\alpha_0^{j'}}{a_0^j} \\
        \mu_2^j &= \hat{\mu}_j \frac{\alpha_2^{j'} \chi_j^2 + \alpha_0^{j'}}{\chi_j^4 + a_2^j \chi_j^2 + a_0^j} \\
        \mu_4^j &= \hat{\mu}_j \frac{\chi_j(\chi_j^2 + \alpha_0^{j''})}{\chi_j^4 + a_2^j \chi_j^2 + a_0^j} (\delta_{ji} - \delta_{je}) \\
        \mu_1^j &= \mu_2^j (2\chi_j) \\
        \mu_3^j &= \mu_4^j (2\chi_j),
    \end{align}
    with the leading prefractor $\hat{\mu}_j$ defined as
    \begin{equation}
        \hat{\mu}_j \equiv \frac{n_j k_B T_j}{\nu_{ji}^M}.
        \label{eqn:mu_hat_e}
    \end{equation}
\end{subequations}

As a note, the term $(\delta_{ji} - \delta_{je})$ evaluates to $1$ for ions and $-1$ for electrons.

In the unmagnetized limit ($\chi_e \rightarrow 0$), the gyroviscous terms $\mathbf{T}_3$ and $\mathbf{T}_4$ vanish, while $\mu_0 = \mu_1 = \mu_2 = \mu$, and the stress tensor becomes
\begin{equation}
    \mathbf{T} = \mu \mathbf{W},
\end{equation}
which is identical to the conventional deviatoric stress tensor.
